cmake_minimum_required(VERSION 3.22)
project(HFT_HPC_C VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(HFT_OPENMP "Build OpenMP C engine" ON)
option(HFT_OPENCL "Build OpenCL C GPU backend" ON)
option(HFT_REALDATA "Enable real-data loader" ON)
if(HFT_REALDATA)
  target_sources(hft_core PRIVATE src/ai/real_dataset.c)
  target_compile_definitions(hft_core PUBLIC HFT_HAVE_REALDATA)
endif()
option(HFT_MPI "Build MPI C worker" ON)
option(HFT_NVIDIA "Build cuBLAS/cuRAND through C APIs" OFF)
option(HFT_GUI "Build native C desktop" ON)
option(HFT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(HFT_TESTS "Build regression tests" ON)
find_package(Threads REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${CMAKE_COMMAND}" -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -DDEST=${CMAKE_CURRENT_BINARY_DIR}/generated/kernels.inc -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS include/hft/types.h include/hft/numeric.h include/hft/simulation.h include/hft/backprop.h kernels/compute.c)
add_library(hft_core STATIC src/ipc/launcher.c src/core/platform.c src/core/config.c src/core/export.c src/analytics/metrics.c src/analytics/linear.c src/ai/models.c src/engines/cpu.c src/engines/opencl.c src/engines/nvidia.c vendor/cJSON.c)
target_include_directories(hft_core PUBLIC include PRIVATE vendor "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(hft_core PUBLIC Threads::Threads)
if(MSVC)
 target_compile_options(hft_core PRIVATE /W4 /fp:strict)
 target_compile_definitions(hft_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
 target_compile_options(hft_core PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter -ffp-contract=off)
 target_compile_definitions(hft_core PUBLIC _POSIX_C_SOURCE=200809L)
 target_link_libraries(hft_core PUBLIC m)
 if(HFT_SANITIZE)
  target_compile_options(hft_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hft_core PUBLIC -fsanitize=address,undefined)
 endif()
endif()
if(WIN32)
 target_link_libraries(hft_core PUBLIC advapi32)
endif()
if(HFT_OPENMP)
 find_package(OpenMP COMPONENTS C)
 if(OpenMP_C_FOUND)
  target_link_libraries(hft_core PUBLIC OpenMP::OpenMP_C)
  target_compile_definitions(hft_core PUBLIC HFT_HAVE_OPENMP)
 endif()
endif()
if(HFT_OPENCL)
 find_package(OpenCL)
 if(OpenCL_FOUND)
  target_link_libraries(hft_core PUBLIC OpenCL::OpenCL)
  target_compile_definitions(hft_core PUBLIC HFT_HAVE_OPENCL)
 endif()
endif()
if(HFT_NVIDIA)
 find_package(CUDAToolkit REQUIRED)
 target_link_libraries(hft_core PUBLIC CUDA::cudart CUDA::cublas CUDA::curand)
 target_compile_definitions(hft_core PUBLIC HFT_HAVE_NVIDIA)
endif()
if(HFT_MPI)
 find_package(MPI COMPONENTS C)
 if(MPI_C_FOUND)
  target_compile_definitions(hft_core PUBLIC HFT_HAVE_MPI)
  add_executable(hft_mpi_worker apps/mpi_worker.c)
  target_link_libraries(hft_mpi_worker PRIVATE hft_core MPI::MPI_C)
  install(TARGETS hft_mpi_worker RUNTIME DESTINATION bin)
 endif()
endif()
add_executable(hft_bench apps/bench.c)
target_include_directories(hft_bench PRIVATE vendor)
target_link_libraries(hft_bench PRIVATE hft_core)
if(HFT_GUI)
 add_library(hft_benchmark STATIC apps/bench.c)
 target_compile_definitions(hft_benchmark PRIVATE HFT_NO_MAIN)
 target_include_directories(hft_benchmark PRIVATE vendor)
 target_link_libraries(hft_benchmark PUBLIC hft_core)
 if(WIN32)
  add_executable(hft_studio WIN32 apps/studio.c)
  target_link_libraries(hft_studio PRIVATE hft_benchmark user32 gdi32 comdlg32 shell32)
 else()
  find_package(X11)
  if(X11_FOUND)
   add_executable(hft_studio apps/studio.c)
   target_link_libraries(hft_studio PRIVATE hft_benchmark X11::X11)
  endif()
 endif()
 if(TARGET hft_studio)
  target_include_directories(hft_studio PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  install(TARGETS hft_studio RUNTIME DESTINATION bin)
 endif()
endif()
if(HFT_TESTS)
 enable_testing()
 add_executable(hft_tests tests/test_main.c)
 target_link_libraries(hft_tests PRIVATE hft_core)
 add_test(NAME core_regression COMMAND hft_tests)
 add_test(NAME benchmark_smoke COMMAND hft_bench --events 100 --symbols 4 --engines serial_naive,serial_opt,threaded --warmup 0 --reps 1 --out smoke)
endif()
install(TARGETS hft_bench RUNTIME DESTINATION bin)
install(DIRECTORY configs docs examples DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
set(CPACK_GENERATOR "ZIP")
set(CPACK_PACKAGE_FILE_NAME "HFT-HPC-C-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}")
include(CPack)
if(HFT_TESTS AND OpenCL_FOUND)
 add_executable(hft_device_tests tests/device_tests.c)
 target_link_libraries(hft_device_tests PRIVATE hft_core)
 add_test(NAME device_regression COMMAND hft_device_tests)
 set_tests_properties(device_regression PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
endif()
option(HFT_EXAMPLES "Build small C teaching examples" ON)
if(HFT_EXAMPLES)
 foreach(EXAMPLE contracts run_engine risk train)
  add_executable(example_${EXAMPLE} examples/${EXAMPLE}.c)
  target_link_libraries(example_${EXAMPLE} PRIVATE hft_core)
 endforeach()
endif()
