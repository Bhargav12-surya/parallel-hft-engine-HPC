cmake_minimum_required(VERSION 3.22)
project(HFT_LAB VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(HFT_ENABLE_OPENMP "Enable genuine OpenMP shared-memory backend" ON)
option(HFT_ENABLE_MPI "Enable MPI distributed-memory backend when installed" ON)
option(HFT_ENABLE_CUDA "Build CUDA feature/inference kernels (requires nvcc)" OFF)
option(HFT_BUILD_TESTS "Build native unit tests" ON)
set(HFT_SANITIZER "" CACHE STRING "Sanitizer: address, undefined or thread; GCC/Clang only")
# Compatibility with the shorter flags used in initial project documentation.
if(DEFINED HFT_OPENMP)
  set(HFT_ENABLE_OPENMP ${HFT_OPENMP})
endif()
if(DEFINED HFT_MPI)
  set(HFT_ENABLE_MPI ${HFT_MPI})
endif()
if(DEFINED HFT_CUDA)
  set(HFT_ENABLE_CUDA ${HFT_CUDA})
endif()
if(HFT_ENABLE_CUDA)
  # Initialize CUDA target defaults before creating the core library.
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
endif()
find_package(Threads REQUIRED)
add_library(hftlab_core STATIC
  native/core/platform.c native/core/market.c native/core/orderbook.c
  native/core/simulation.c native/core/results.c native/ai/inference.c)
target_include_directories(hftlab_core PUBLIC native/include)
target_link_libraries(hftlab_core PUBLIC Threads::Threads)
if(MSVC)
  if(HFT_SANITIZER)
    message(FATAL_ERROR "These sanitizer presets require GCC/Clang on Linux/WSL; no sanitizer was enabled for MSVC")
  endif()
  target_compile_options(hftlab_core PRIVATE $<$<COMPILE_LANGUAGE:C>:/W4> $<$<COMPILE_LANGUAGE:C>:/fp:strict>)
  target_compile_definitions(hftlab_core PUBLIC _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(hftlab_core PUBLIC psapi)
else()
  target_compile_options(hftlab_core PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-ffp-contract=off>)
  target_compile_definitions(hftlab_core PUBLIC _POSIX_C_SOURCE=200809L)
  target_link_libraries(hftlab_core PUBLIC m)
  if(HFT_SANITIZER)
    if(NOT HFT_SANITIZER MATCHES "^(address|undefined|thread)$")
      message(FATAL_ERROR "Choose one supported sanitizer: address, undefined, thread")
    endif()
    if(HFT_SANITIZER STREQUAL "thread" AND HFT_ENABLE_OPENMP)
      message(FATAL_ERROR "Use -DHFT_ENABLE_OPENMP=OFF for the separate ThreadSanitizer build")
    endif()
    target_compile_options(hftlab_core PUBLIC -fsanitize=${HFT_SANITIZER} -fno-omit-frame-pointer)
    target_link_options(hftlab_core PUBLIC -fsanitize=${HFT_SANITIZER})
  endif()
endif()
if(HFT_ENABLE_OPENMP)
  find_package(OpenMP QUIET COMPONENTS C)
  if(OpenMP_C_FOUND)
    target_link_libraries(hftlab_core PUBLIC OpenMP::OpenMP_C)
    target_compile_definitions(hftlab_core PUBLIC HFT_HAVE_OPENMP)
  else()
    message(STATUS "OpenMP unavailable; CPU engines remain usable")
  endif()
endif()
if(HFT_ENABLE_MPI)
  find_package(MPI QUIET COMPONENTS C)
  if(MPI_C_FOUND)
    target_link_libraries(hftlab_core PUBLIC MPI::MPI_C)
    target_compile_definitions(hftlab_core PUBLIC HFT_HAVE_MPI)
  else()
    message(STATUS "MPI unavailable; distributed backend disabled")
  endif()
endif()
if(HFT_ENABLE_CUDA)
  target_sources(hftlab_core PRIVATE native/engines/cuda/compute.cu)
  target_link_libraries(hftlab_core PUBLIC CUDA::cudart)
  target_compile_definitions(hftlab_core PUBLIC HFT_HAVE_CUDA)
  target_compile_options(hftlab_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
else()
  target_sources(hftlab_core PRIVATE native/engines/cuda/stub.c)
endif()
add_executable(hftlab_runner native/runners/main.c)
target_link_libraries(hftlab_runner PRIVATE hftlab_core)
if(HFT_BUILD_TESTS)
  enable_testing()
  add_executable(hftlab_native_tests native/tests/test_core.c)
  target_link_libraries(hftlab_native_tests PRIVATE hftlab_core)
  add_test(NAME native_orderbook_accounting_features COMMAND hftlab_native_tests)
  add_test(NAME native_capability_protocol COMMAND hftlab_runner --capabilities)
endif()
install(TARGETS hftlab_runner RUNTIME DESTINATION bin)
